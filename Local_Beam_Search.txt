import random

dist = [
    [0, 10, 15, 20],
    [10, 0, 35, 25],
    [15, 35, 0, 30],
    [20, 25, 30, 0]
]

n = 4
k = 2

def distance(path):
    total = 0
    for i in range(n):
        total += dist[path[i]][path[(i+1) % n]]
    return total

# Generate initial routes
beam = []
for i in range(k):
    path = list(range(n))
    random.shuffle(path)
    beam.append(path)

# Local Beam Search
for step in range(100):
    new_routes = []

    for path in beam:
        for i in range(1, n):
            new = path[:]
            new[0], new[i] = new[i], new[0]
            new_routes.append(new)

    new_routes.sort(key=distance)
    beam = new_routes[:k]

# Best route
best = min(beam, key=distance)

print("Best Route:", best + [best[0]])
print("Minimum Distance:", distance(best))
