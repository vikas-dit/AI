import random

# Distance matrix
dist = [
    [0, 10, 15, 20],
    [10, 0, 35, 25],
    [15, 35, 0, 30],
    [20, 25, 30, 0]
]

n = 4
k = 2

# Calculate route distance
def distance(path):
    total = 0
    for i in range(n):
        total += dist[path[i]][path[(i+1) % n]]
    return total

# Local Beam Search
beam = []

# Generate initial routes
for i in range(k):
    path = list(range(n))
    random.shuffle(path)
    beam.append(path)

# Search for better routes
for step in range(100):

    new_routes = []

    for path in beam:
        for i in range(1, n):
            new_path = path[:]
            new_path[0], new_path[i] = new_path[i], new_path[0]
            new_routes.append(new_path)

    new_routes.sort(key=distance)

    best_routes = new_routes[:k]

    if distance(best_routes[0]) >= distance(beam[0]):
        break

    beam = best_routes

# Display result
best = min(beam, key=distance)

print("Best Route:", best + [best[0]])
print("Minimum Distance:", distance(best))
