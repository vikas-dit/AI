from collections import deque

start = (1, 2, 3, 4, 0, 6, 7, 5, 8)
goal = (1, 2, 3, 4, 5, 6, 7, 8, 0)

def bfs(start):
    queue = deque([[start]])
    visited = {start}

    while queue:
        path = queue.popleft()
        state = path[-1]

        if state == goal:
            return path

        zero = state.index(0)

        for move in [-3, 3, -1, 1]:
            pos = zero + move

            if 0 <= pos < 9:
                if move == -1 and zero % 3 == 0:
                    continue
                if move == 1 and zero % 3 == 2:
                    continue

                new = list(state)
                new[zero], new[pos] = new[pos], new[zero]
                new = tuple(new)

                if new not in visited:
                    visited.add(new)
                    queue.append(path + [new])

    return None

solution = bfs(start)

if solution:
    for state in solution:
        print(state[:3])
        print(state[3:6])
        print(state[6:])
        print()
    print("Total Moves:", len(solution) - 1)
else:
    print("No Solution")
