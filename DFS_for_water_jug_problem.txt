# Water Jug Problem using DFS

jug1 = 4
jug2 = 3
target = 2

visited = set()

def dfs(a, b):
    if a == target or b == target:
        print(a, b)
        return True

    if (a, b) in visited:
        return False

    visited.add((a, b))
    print(a, b)

    operations = [
        (jug1, b), (a, jug2),  # Fill
        (0, b), (a, 0),        # Empty
        (a-min(a, jug2-b), b+min(a, jug2-b)), # Pour 1 to 2
        (a+min(b, jug1-a), b-min(b, jug1-a))  # Pour 2 to 1
    ]

    for x, y in operations:
        if dfs(x, y):
            return True

    return False

if not dfs(0, 0):
    print("Solution not found")
